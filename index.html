-- Governed Semantic Query Generated by Semantic Parser v4.2
SELECT
s.supplier_id, s.supplier_name, s.country_code,
COUNT(sh.shipment_id) AS delayed_shipment_count,
ROUND(AVG(sh.delay_hours), 1) AS avg_delay_hours,
ROUND(AVG(po.lead_time_days), 1) AS avg_lead_time_days,
ROUND((COUNT(CASE WHEN sh.is_on_time = 1 THEN 1 END) * 100.0) / COUNT(sh.shipment_id), 1) AS otif_score,
r.primary_root_cause
FROM ontology.Supplier s
JOIN ontology.PurchaseOrder po ON s.supplier_id = po.vendor_id
JOIN ontology.Shipment sh ON po.po_number = sh.po_reference_id
LEFT JOIN ontology.DelayReason r ON sh.shipment_id = r.shipment_id
WHERE sh.delivery_date BETWEEN '2024-07-01' AND '2024-09-30'
AND sh.delay_hours > 0
GROUP BY s.supplier_id, s.supplier_name, s.country_code, r.primary_root_cause
ORDER BY delayed_shipment_count DESC;
